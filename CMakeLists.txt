cmake_minimum_required(VERSION 3.24)
project(cuda_knn_cpp VERSION 1.0.0 LANGUAGES CXX)
option(KNN_CPU_ONLY "Build CPU reference executable for non-CUDA tests only" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Threads REQUIRED)
find_path(JSONC_INCLUDE_DIR json-c/json.h REQUIRED)
find_library(JSONC_LIBRARY NAMES json-c REQUIRED)
find_path(BOOST_ASIO_HEADER boost/asio.hpp REQUIRED)
add_executable(knn_server src/main.cpp src/index.cpp)
target_include_directories(knn_server PRIVATE include ${JSONC_INCLUDE_DIR} ${BOOST_ASIO_HEADER})
target_compile_definitions(knn_server PRIVATE BOOST_ERROR_CODE_HEADER_ONLY)
target_compile_options(knn_server PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-O3> $<$<COMPILE_LANGUAGE:CXX>:-Wall> $<$<COMPILE_LANGUAGE:CXX>:-Wextra>)
target_link_libraries(knn_server PRIVATE ${JSONC_LIBRARY} Threads::Threads)
if(KNN_CPU_ONLY)
  target_sources(knn_server PRIVATE src/cpu_backend.cpp)
  message(WARNING "CPU_REFERENCE_TEST_ONLY selected; this executable is not the CUDA implementation")
else()
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  set(CMAKE_CUDA_STANDARD 17)
  set(CMAKE_CUDA_STANDARD_REQUIRED ON)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES "75-real;80-real;86-real;89-real;90-real;120-real;89-virtual")
  endif()
  target_sources(knn_server PRIVATE src/cuda_backend.cu)
  set_target_properties(knn_server PROPERTIES CUDA_SEPARABLE_COMPILATION OFF CUDA_ARCHITECTURES "${CMAKE_CUDA_ARCHITECTURES}")
  target_compile_options(knn_server PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3>)
  target_link_libraries(knn_server PRIVATE CUDA::cudart CUDA::cublas)
endif()
